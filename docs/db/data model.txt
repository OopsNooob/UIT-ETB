@startuml
' Cấu hình giao diện chuẩn cho ERD
hide circle
skinparam linetype ortho
skinparam class {
    BackgroundColor White
    BorderColor DarkSlateGray
    ArrowColor DarkSlateGray
}

' =========================================
' CORE ENTITIES
' =========================================
entity "users" as users {
  * id : UUID <<PK>>
  --
  name : VARCHAR
  email : VARCHAR <<UNIQUE>>
  phone : VARCHAR
  role : ENUM("admin", "organizer", "user")
  password_hash : VARCHAR
  is_active : BOOLEAN
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
  deleted_at : TIMESTAMP
}

entity "events" as events {
  * id : UUID <<PK>>
  --
  organizer_id : UUID <<FK>>
  title : VARCHAR
  description : TEXT
  location : VARCHAR
  banner_url : VARCHAR
  start_date : TIMESTAMP
  end_date : TIMESTAMP
  total_capacity : INT
  remaining_capacity : INT
  status : ENUM("active", "cancelled", "completed")
  created_by : UUID <<FK>>
  updated_by : UUID <<FK>>
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
  deleted_at : TIMESTAMP
}

' =========================================
' TICKET INVENTORY & SALES
' =========================================
entity "ticket_types" as ticket_types {
  * id : UUID <<PK>>
  --
  event_id : UUID <<FK>>
  name : VARCHAR
  description : TEXT
  price : DECIMAL
  total_quantity : INT
  sold_quantity : INT
  sale_start : TIMESTAMP
  sale_end : TIMESTAMP
  created_by : UUID <<FK>>
  updated_by : UUID <<FK>>
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
  deleted_at : TIMESTAMP
}

entity "purchases" as purchases {
  * id : UUID <<PK>>
  --
  user_id : UUID <<FK>>
  event_id : UUID <<FK>>
  gateway_transaction_id : VARCHAR
  total_amount : DECIMAL
  discount_amount : DECIMAL
  final_amount : DECIMAL
  status : ENUM("pending", "completed", "failed", "refunded")
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
  deleted_at : TIMESTAMP
}

entity "tickets" as tickets {
  * id : UUID <<PK>>
  --
  purchase_id : UUID <<FK>> (Nullable for manual issue)
  event_id : UUID <<FK>>
  ticket_type_id : UUID <<FK>>
  user_id : UUID <<FK>> (Current Owner)
  ticket_code : VARCHAR <<UNIQUE>> (QR String)
  status : ENUM("valid", "used", "cancelled", "refunded")
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
  deleted_at : TIMESTAMP
}

' =========================================
' PROMOTIONS & QUEUES
' =========================================
entity "giftcodes" as giftcodes {
  * id : UUID <<PK>>
  --
  code : VARCHAR <<UNIQUE>>
  type : ENUM("fixed_amount", "percentage")
  discount_value : DECIMAL
  event_id : UUID <<FK>> (Nullable for global code)
  usage_limit : INT
  used_count : INT
  expired_date : TIMESTAMP
  status : ENUM("active", "expired", "depleted")
  created_by : UUID <<FK>>
  updated_by : UUID <<FK>>
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
  deleted_at : TIMESTAMP
}

entity "giftcode_redemptions" as giftcode_redemptions {
  * id : UUID <<PK>>
  --
  giftcode_id : UUID <<FK>>
  user_id : UUID <<FK>>
  purchase_id : UUID <<FK>>
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
  deleted_at : TIMESTAMP
}

entity "waiting_list" as waiting_list {
  * id : UUID <<PK>>
  --
  user_id : UUID <<FK>>
  event_id : UUID <<FK>>
  status : ENUM("waiting", "offered", "expired", "purchased")
  offer_expires_at : TIMESTAMP
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
  deleted_at : TIMESTAMP
}

' =========================================
' SECURITY & AUDITING
' =========================================
entity "audit_logs" as audit_logs {
  * id : UUID <<PK>>
  --
  user_id : UUID <<FK>> (Nullable)
  action : VARCHAR
  target_entity : VARCHAR
  target_id : VARCHAR
  ip_address : VARCHAR
  details : TEXT <<JSON>>
  event_code : VARCHAR(10)
  security_level : VARCHAR
  created_at : TIMESTAMP
}

' =========================================
' RELATIONSHIPS
' =========================================
users ||--o{ events : organizes
users ||--o{ purchases : makes
users ||--o{ tickets : owns
users ||--o{ waiting_list : joins
users ||--o{ audit_logs : performs
users ||--o{ giftcode_redemptions : redeems

events ||--|{ ticket_types : configures
events ||--o{ tickets : issues
events ||--o{ purchases : generates
events ||--o{ giftcodes : specific to
events ||--o{ waiting_list : has squeues

ticket_types ||--|{ tickets : instantiates

purchases ||--|{ tickets : contains
purchases ||--o| giftcode_redemptions : applies

giftcodes ||--o{ giftcode_redemptions : tracks usage

@enduml
